Registar.php
<?php
include("conexion.php");

$id_usuario       = $_POST['id_usuario'];
$id_cliente       = $_POST['id_cliente'];
$tipo_mas_vendido = mysqli_real_escape_string($conne, $_POST['tipo_mas_vendido']);
$cant_total       = $_POST['cant_total'];
$total            = $_POST['total'];
$metodo_pago      = mysqli_real_escape_string($conne, $_POST['metodo_pago']);
$estado           = mysqli_real_escape_string($conne, $_POST['estado']);

if (
    $id_usuario === '' || $cant_total === '' || $total === '' || $metodo_pago === '' || $estado === ''
    || !is_numeric($id_usuario) || !is_numeric($cant_total) || !is_numeric($total)
    || ($id_cliente !== '' && !is_numeric($id_cliente))
) {
    die("Datos inválidos");
} else {
    if ($id_cliente == '') {
        $id_cliente = 'NULL';
    }

    $sql  = "INSERT INTO ventas (id_usuario, id_cliente, cant_total, tipo_mas_vendido, total, metodo_pago, estado) VALUES ($id_usuario, $id_cliente, $cant_total, '$tipo_mas_vendido', $total, '$metodo_pago', '$estado')";
    if (mysqli_query($conne, $sql)) {

        header("Location: index.php?mensaje=registrado");
        exit();
    } else {
        echo "error al registrar: ", mysqli_error($conne);
    }
}
